```tsx
// src/screens/AssessmentScreen.tsx
import { useState } from 'react';
import { DIMENSIONS, type DimensionKey } from '../config/dimensions';
import { saveAssessment } from '../data/assessments';
import { Slider } from '@/components/ui/slider';
import { Button } from '@/components/ui/button';

interface AssessmentScreenProps {
  onComplete: () => void;
}

// Scores keyed by DimensionKey — persists across step navigation (D-06)
type Scores = Record<DimensionKey, number>;

function buildInitialScores(): Scores {
  return Object.fromEntries(
    DIMENSIONS.map((d) => [d.key, 50])
  ) as Scores;
}

export function AssessmentScreen({ onComplete }: AssessmentScreenProps) {
  const [step, setStep] = useState(0);
  const [scores, setScores] = useState<Scores>(buildInitialScores);
  const [saving, setSaving] = useState(false);
  const [saveError, setSaveError] = useState(false);
  // direction: 'forward' | 'back' — used to pick slide transition direction
  const [direction, setDirection] = useState<'forward' | 'back'>('forward');

  const current = DIMENSIONS[step];
  const isFirst = step === 0;
  const isLast = step === DIMENSIONS.length - 1;
  const currentScore = scores[current.key];

  function handleSliderChange(values: number[]) {
    setScores((prev) => ({ ...prev, [current.key]: values[0] }));
  }

  function handleBack() {
    if (!isFirst) {
      setDirection('back');
      setStep((s) => s - 1);
      setSaveError(false);
    }
  }

  async function handleNext() {
    if (saving) return;
    setSaveError(false);

    if (isLast) {
      setSaving(true);
      try {
        await saveAssessment(scores);
        onComplete();
      } catch {
        setSaveError(true);
        setSaving(false);
      }
    } else {
      setDirection('forward');
      setStep((s) => s + 1);
    }
  }

  return (
    <div className="min-h-screen bg-white flex flex-col">
      {/* App bar */}
      <header className="h-14 flex items-center px-8 border-b border-slate-200">
        <span className="text-[28px] font-semibold text-slate-900 leading-[1.2]">
          Peace Index
        </span>
      </header>

      {/* Main content — centered column, max 560px */}
      <main className="flex-1 flex flex-col items-center px-8 pt-8">
        <div className="w-full max-w-[560px] flex flex-col gap-6">

          {/* Progress indicator — D-05: "Step N of 5" */}
          <div className="flex flex-col items-center gap-2">
            <span className="text-sm text-slate-500">
              Step {step + 1} of {DIMENSIONS.length}
            </span>
            {/* 5-dot progress */}
            <div className="flex gap-2">
              {DIMENSIONS.map((d, i) => (
                <div
                  key={d.key}
                  className={[
                    'w-2 h-2 rounded-full transition-colors duration-200',
                    i < step
                      ? 'bg-blue-500/60'         // completed: accent at 60% opacity
                      : i === step
                        ? 'bg-blue-500'           // active: full accent
                        : 'bg-slate-200',         // future: border color
                  ].join(' ')}
                />
              ))}
            </div>
          </div>

          {/* Dimension card */}
          <div className="rounded-lg border border-slate-200 bg-white p-6 flex flex-col gap-6">

            {/* Dimension name — Heading: 20px/600 */}
            <h2 className="text-[20px] font-semibold text-slate-900 leading-[1.2]">
              {current.label}
            </h2>

            {/* Dimension description — Body: 16px/400 */}
            <p className="text-[16px] text-slate-500 leading-[1.5]">
              {current.description}
            </p>

            {/* Slider + numeric readout — D-01, D-02, D-03 */}
            <div className="flex flex-col gap-4">
              {/* Numeric readout above slider — always visible, never under thumb (D-03) */}
              <div className="flex items-center justify-between">
                <span className="text-sm text-slate-500">{current.label}</span>
                <span className="text-[28px] font-semibold text-blue-500 leading-[1.2]">
                  {currentScore}%
                </span>
              </div>

              {/* shadcn Slider: continuous 1-100 (D-02), step=1 for integer precision */}
              <Slider
                min={1}
                max={100}
                step={1}
                value={[currentScore]}
                onValueChange={handleSliderChange}
                className="w-full"
              />

              {/* Scale labels */}
              <div className="flex justify-between text-xs text-slate-400">
                <span>1%</span>
                <span>100%</span>
              </div>
            </div>
          </div>

          {/* Save error message */}
          {saveError && (
            <p className="text-sm text-red-500 text-center">
              Your assessment couldn't be saved. Please try again.
            </p>
          )}

          {/* Navigation row — D-06: back navigation */}
          <div className="flex gap-3">
            <Button
              variant="outline"
              onClick={handleBack}
              disabled={isFirst}
              className="flex-1"
            >
              Back
            </Button>
            <Button
              onClick={handleNext}
              disabled={saving}
              className="flex-1 bg-blue-500 hover:bg-blue-600 text-white"
            >
              {saving ? 'Saving…' : isLast ? 'Save Assessment' : 'Next'}
            </Button>
          </div>

        </div>
      </main>
    </div>
  );
}
```

**Key implementation notes:**

1. Transitions (D-04, UI-SPEC): The `direction` state is set but simple CSS transitions are sufficient in Phase 1. If the CSS transition classes need to be wired up for the slide-left/slide-right effect, wrap the dimension card in a container with these Tailwind classes applied via `key={step}` to force React to remount on step change:
   - Forward: outgoing slides left, incoming slides from right
   - Use `key={step}` on the card container to trigger React remount and CSS animation
   - Implementation: Add `animate-in fade-in slide-in-from-right-4 duration-200` to the card div when moving forward
   - If shadcn@canary does not include animate-in utilities, fall back to `transition-opacity duration-200` — do not add a third-party animation library

2. The `saving` state prevents double-submits on slow devices.

3. `buildInitialScores` is a function reference (not inline arrow) to avoid recreating the initial state object on every render.

4. The error UI uses the exact copy from UI-SPEC.md Copywriting Contract.
